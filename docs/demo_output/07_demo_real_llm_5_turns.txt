```python
import tweepy

class StockTweetListener(tweepy.StreamingClient):
    def __init__(self, bearer_token, keywords, tweet_callback):
        super().__init__(bearer_token)
        self.keywords = keywords
        self.tweet_callback = tweet_callback
        self._setup_rules()

    def _setup_rules(self):
        # Remove existing rules
        try:
            rules = self.get_rules().data
            if rules:
                rule_ids = [rule.id for rule in rules]
                self.delete_rules(rule_ids)
        except Exception as e:
            print(f"Error fetching/deleting rules: {e}")

        # Add new rules for filtering
        for kw in self.keywords:
            try:
                self.add_rules(tweepy.StreamRule(value=kw))
            except Exception as e:
                print(f"Error adding rule for {kw}: {e}")

    # This method is called when a tweet is received
    def on_tweet(self, tweet):
        # Process the tweet using the provided callback function
        self.tweet_callback(tweet)

    def on_errors(self, errors):
        print(f"Stream encountered errors: {errors}")

    def on_exception(self, exception):
        print(f"Stream encountered exception: {exception}")

# Example tweet processing callback
def process_tweet(tweet):
    print(f"Tweet received: {tweet.text}")

if __name__ == "__main__":
    import os

    # Load your Twitter Bearer Token securely, e.g. environment variable
    BEARER_TOKEN = os.getenv("TWITTER_BEARER_TOKEN")
    if not BEARER_TOKEN:
        raise RuntimeError("Twitter Bearer Token is not set in environment variables")

    # Keywords to track (stocks symbols usually prefixed with $)
    stock_keywords = ["$AAPL", "$TSLA", "$AMZN"]

    # Create stream listener instance

    # Start streaming (runs indefinitely)
```

```python
from vaderSentiment.vaderSentiment import SentimentIntensityAnalyzer

# Initialize the VADER sentiment analyzer once
analyzer = SentimentIntensityAnalyzer()

def get_tweet_sentiment(tweet_text):
    """
    Analyzes the sentiment of a tweet text using VADER.

    Args:
        tweet_text (str): The text of the tweet.

    Returns:
        dict: Dictionary with 'positive', 'neutral', 'negative', and 'compound' sentiment scores.
    """
    # Compute sentiment scores
    sentiment = analyzer.polarity_scores(tweet_text)
    # sentiment is a dict like:
    # {'neg': 0.0, 'neu': 0.625, 'pos': 0.375, 'compound': 0.6249}

    # Return scores with more descriptive keys
    return {
        'positive': sentiment['pos'],
        'neutral': sentiment['neu'],
        'negative': sentiment['neg'],
        'compound': sentiment['compound']
    }

# Example usage:
if __name__ == "__main__":
    example_text = "I am very bullish on $AAPL stock! Great prospects ahead."
    scores = get_tweet_sentiment(example_text)
    print(f"Sentiment scores: {scores}")
```

Explanation:
- The function uses the `SentimentIntensityAnalyzer` from `vaderSentiment` to get sentiment scores.
- `compound` is a normalized score that summarizes the overall sentiment from -1 (most negative) to +1 (most positive).
- Positive, neutral, and negative scores correspond to the proportions of the text classified into those sentiments.

   - Define thresholds (e.g., max loss percentage).

   - Check if the current loss exceeds stop-loss limit.
4. **Trigger Sell Orders:**
   - When stop-loss is breached, execute a sell order via trading API.

Below is a Python class implementation of this module. It assumes integration with an Alpaca API client for executing orders.

```python
import logging

class RiskManager:
    def __init__(self, alpaca_api_client, stop_loss_pct=0.05):
        """
        Initialize Risk Manager.

        Args:
            alpaca_api_client: Instance of alpaca_trade_api.REST.
            stop_loss_pct (float): Stop loss threshold as decimal (e.g. 0.05 = 5% loss).
        """
        self.api = alpaca_api_client
        self.stop_loss_pct = stop_loss_pct
        self.open_positions = {}  # key: symbol, value: dict with 'qty', 'entry_price'

        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(__name__)

    def update_positions(self):
        """
        Fetch current open positions from Alpaca and update internal tracking.
        """
        positions = self.api.list_positions()
        self.open_positions.clear()
        for pos in positions:
            self.open_positions[pos.symbol] = {
                'qty': float(pos.qty),
                'entry_price': float(pos.avg_entry_price)
            }
        self.logger.info(f"Updated open positions: {list(self.open_positions.keys())}")

    def check_stop_loss(self):
        """
        Check all open positions for stop loss breach.
        If breached, trigger sell order.
        """
        for symbol, pos in self.open_positions.items():
            try:
                # Get current market price for symbol
                barset = self.api.get_latest_trade(symbol)
                current_price = float(barset.price)
            except Exception as e:
                self.logger.error(f"Failed to fetch price for {symbol}: {e}")
                continue

            entry_price = pos['entry_price']
            qty = pos['qty']
            loss_pct = (entry_price - current_price) / entry_price

            self.logger.info(f"Symbol: {symbol} | Entry Price: {entry_price} | Current Price: {current_price} | Loss%: {loss_pct:.4f}")

            # Check if loss exceeds threshold
            if loss_pct >= self.stop_loss_pct:
                self.logger.warning(f"Stop loss triggered for {symbol}: Loss = {loss_pct:.2%}. Selling {qty} shares.")
                self.trigger_sell_order(symbol, qty)

    def trigger_sell_order(self, symbol, qty):
        """
        Execute a sell order for the given symbol and quantity.

        Args:
            symbol (str): Stock ticker symbol.
            qty (float): Quantity to sell.
        """
        try:
            order = self.api.submit_order(
                symbol=symbol,
                qty=qty,
                side='sell',
                type='market',
                time_in_force='gtc'
            )
            self.logger.info(f"Submitted sell order for {qty} shares of {symbol}. Order ID: {order.id}")
        except Exception as e:
            self.logger.error(f"Failed to submit sell order for {symbol}: {e}")

# Example usage (with Alpaca API client properly instantiated):
if __name__ == "__main__":
    import os
    import alpaca_trade_api as tradeapi

    # Setup Alpaca API credentials via env variables
    ALPACA_API_KEY = os.getenv("ALPACA_API_KEY")
    ALPACA_SECRET_KEY = os.getenv("ALPACA_SECRET_KEY")
    ALPACA_BASE_URL = "https://paper-api.alpaca.markets"

    api = tradeapi.REST(ALPACA_API_KEY, ALPACA_SECRET_KEY, ALPACA_BASE_URL, api_version='v2')

    risk_manager = RiskManager(api, stop_loss_pct=0.05)  # 5% stop loss

    risk_manager.update_positions()
    risk_manager.check_stop_loss()
```
